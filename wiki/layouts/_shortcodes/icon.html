{{/* Inline icon. Compatible with the former Wowchemy `icon` shortcode, which used Font Awesome. Font Awesome names are mapped to the HugoBlox (Heroicons) icon set. Parameters ---------- name : Icon name (Font Awesome name or HugoBlox icon name, e.g. "hero/document"). pack : optional Former Font Awesome pack ("fas", "far", "fab"). Ignored. padding_left : default "0" Left padding. padding_right : default "1" Right padding. */}} {{- if (.Get "name") -}} {{- $fa := dict "file-pdf" "document-text" "file-code" "code-bracket" "file" "document" "download" "arrow-down-tray" "info-circle" "information-circle" -}} {{- $icon := index $fa (.Get "name") | default (.Get "name") -}} {{- $padding_right := (.Get "padding_right") | default 1 -}} {{- partial "functions/get_icon" (dict "name" $icon "attributes" "style=\"height: 1em; transform: translateY(0.1em);\"") -}} {{- end -}}
